32/// Docker-based Speculos driver
33pub struct DockerDriver {
34    d: Docker,
35}
36
37/// Handle to a Speculos instance running under Docker
38#[derive(Debug)]
39pub struct DockerHandle {
40    name: String,
41    addr: SocketAddr,
42    exit_tx: Sender<()>,
43}
44
45impl DockerDriver {
46    /// Create a new docker driver
47    pub fn new() -> Result<Self, anyhow::Error> {
48        // Connect to docker instance
49        let d = Docker::connect_with_local_defaults()?;
50
51        // Return driver
52        Ok(Self { d })
53    }
54}
55
56const DEFAULT_IMAGE: &str = "ghcr.io/ledgerhq/ledger-app-builder/ledger-app-dev-tools:latest";
57
58/// [Driver] implementation for [DockerDriver]
59#[async_trait]
60impl Driver for DockerDriver {
61    type Handle = DockerHandle;
62
63    async fn run(&self, app: &str, opts: Options) -> anyhow::Result<Self::Handle> {
64        // Set container name
65        let name = format!(
66            "speculos-{}-{}",
67            opts.model.to_string().to_lowercase(),
68            opts.http_port
69        );
70
71        let create_options = CreateContainerOptionsBuilder::default().name(&name).build();
72
73        // Setup ports
74        let mut ports = vec![opts.http_port];
75        if let Some(p) = opts.apdu_port {
76            ports.push(p);
77        }
78
79        let exposed_ports = ports.iter().map(|p| {
80            let b = PortBinding {
81                host_port: Some(format!("{p}/tcp")),
82                ..Default::default()
83            };
84            (format!("{p}/tcp"), vec![b], HashMap::<(), ()>::new())
85        });
86
87        let app_path = PathBuf::from(app);
88        let app_file = app_path.file_name().and_then(|n| n.to_str()).unwrap();
89
90        // Setup speculos command
91
92        let mut cmd = vec!["speculos".to_string()];
93        cmd.append(&mut opts.args());
94        cmd.push(format!("/app/{app_file}"));
95
96        debug!("command: {}", cmd.join(" "));
97
98        // Setup container
99        let create_config = ContainerCreateBody {
100            image: Some(DEFAULT_IMAGE.to_string()),
101            cmd: Some(cmd),
102            attach_stdout: Some(true),
103            attach_stderr: Some(true),
104            stop_signal: Some("KILL".to_string()),
105            exposed_ports: Some(HashMap::from_iter(
106                exposed_ports.clone().map(|p| (p.0, p.2)),
107            )),
108            host_config: Some(HostConfig {
109                #[cfg(target_os = "linux")]
110                binds: Some(vec![String::from("/tmp/.X11-unix:/tmp/.X11-unix")]),
111                port_bindings: Some(HashMap::from_iter(exposed_ports.map(|p| (p.0, Some(p.1))))),
112                ..Default::default()
113            }),
114
115            env: {
116                #[cfg(any(target_os = "windows", target_os = "macos"))]
117                let display = Some(vec![String::from("DISPLAY=host.docker.internal:0")]);
118                #[cfg(target_os = "linux")]
119                let display = std::env::var("DISPLAY")
120                    .ok()
121                    .map(|d| vec![format!("DISPLAY={}", d)]);
122                display
123            },
124            ..Default::default()
125        };
126
127        // Remove existing container if there is one
128        let remove_options = RemoveContainerOptionsBuilder::new().force(true).build();
129        let _ = self.d.remove_container(&name, Some(remove_options)).await;
130
131        // Create container
132        debug!("Creating container {}", name);
133        let _create_info = self
134            .d
135            .create_container(Some(create_options), create_config)
136            .await?;
137
138        // Generate application archive
139        let mut buff = BytesMut::new();
140        let mut tar = tar::Builder::new((&mut buff).writer());
141
142        tar.append_path_with_name(&app_path, format!("app/{app_file}"))?;
143
144        tar.finish()?;
145        drop(tar);
146
147        // Write app archive to container
148        let upload_options = UploadToContainerOptionsBuilder::new().path("/").build();
149
150        self.d
151            .upload_to_container(&name, Some(upload_options), body_full(buff.to_vec().into()))
152            .await?;
153
154        // Start container
155        debug!("Starting container {}", name);
156        let start_options = StartContainerOptionsBuilder::new().build();
157        let _start_info = self.d.start_container(&name, Some(start_options)).await?;
158
159        debug!("Container started");
160
161        let (exit_tx, mut exit_rx) = channel();
162
163        // Setup log streaming task
164        let log_options = LogsOptionsBuilder::new()
165            .stderr(true)
166            .stdout(true)
167            .follow(true)
168            .build();
169
170        let mut logs = self.d.logs(&name, Some(log_options));
171
172        tokio::task::spawn(async move {
173            debug!("start log task");
174
175            loop {
176                tokio::select! {
177                    // Fetch log entries
178                    l = logs.next() => {
179                        match l {
180                            Some(Ok(v)) => print!("{v}"),
181                            Some(Err(e)) => {
182                                debug!("exit log task: {:?}", e);
183                                break;
184                            },
185                            _ => continue,
186                        }
187                    },
188                    // Handle exit signal
189                    _ = &mut exit_rx => {
190                        break;
191                    }
192                }
193            }
194        });
195
196        // Return container handle
197        let addr = SocketAddr::new(IpAddr::V4(Ipv4Addr::new(127, 0, 0, 1)), opts.http_port);
198        Ok(DockerHandle {
199            name,
200            addr,
201            exit_tx,
202        })
203    }
204
205    async fn wait(&self, handle: &mut Self::Handle) -> anyhow::Result<()> {
206        use ContainerStateStatusEnum::*;
207
208        debug!("Awaiting container completion");
209
210        // Poll container info periodically
211        loop {
212            let inspect_options = InspectContainerOptionsBuilder::new().build();
213
214            // Fetch container info
215            let info = self
216                .d
217                .inspect_container(&handle.name, Some(inspect_options))
218                .await?;
219
220            // Return when container exits
221            match info.state.and_then(|s| s.status) {
222                Some(CREATED) | Some(RUNNING) => (),
223                Some(_) => return Ok(()),
224                _ => (),
225            }
226
227            // Sleep for a while
228            tokio::time::sleep(Duration::from_secs(1)).await;
229        }
230    }
231
232    async fn exit(&self, handle: Self::Handle) -> anyhow::Result<()> {
233        // Stop container
234        debug!("Stopping container {}", handle.name);
235
236        // Send exit signal to log task
237        let _ = handle.exit_tx.send(());
238
239        // Send container stop signal
240        let stop_options = StopContainerOptionsBuilder::new().t(0).build();
241        let options = Some(stop_options);
242        let _ = self.d.stop_container(&handle.name, options).await;
243
244        // Remove container
245        debug!("Removing container");
246        let remove_options = RemoveContainerOptionsBuilder::new().force(true).build();
247        self.d
248            .remove_container(&handle.name, Some(remove_options))
249            .await?;
250
251        debug!("Container removed");
252
253        Ok(())
254    }
255}
256
257#[async_trait]
258impl Handle for DockerHandle {
259    fn addr(&self) -> SocketAddr {
260        self.addr
261    }
