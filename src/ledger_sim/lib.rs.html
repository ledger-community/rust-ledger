1//! Rust wrapper for executing Speculos via local install or docker image,
2//! provided to simplify CI/CD with ledger applications.
3//!
4//! Drivers are provided for [Docker](DockerDriver) and [Local](LocalDriver)
5//! execution, with a [Generic](GenericDriver) abstraction to support
6//! runtime driver selection.
7//!
8//! ### Examples:
9//!
10//! ``` no_run
11//! # use tracing::{debug};
12//! use ledger_sim::{GenericDriver, DriverMode, Driver, Model, Options};
13//! use ledger_lib::{Device, transport::{Transport, TcpTransport, TcpInfo}, DEFAULT_TIMEOUT};
14//! use ledger_proto::apdus::{AppInfoReq, AppInfoResp};
15//!
16//! #[tokio::main]
17//! async fn main() -> anyhow::Result<()> {
18//!     // Setup driver for speculos connection
19//!     let driver = GenericDriver::new(DriverMode::Docker)?;
20//!
21//!     // Launch speculos with the provided app
22//!     let opts = Options {
23//!         model: Model::NanoX,
24//!         apdu_port: Some(1237),
25//!         ..Default::default()
26//!     };
27//!     let mut handle = driver.run("ledger-app", opts).await?;
28//!
29//!     // Setup TCP APDU transport to speculos
30//!     let mut transport = TcpTransport::new()?;
31//!     let mut device = transport.connect(TcpInfo::default()).await?;
32//!
33//!     // Fetch app info via transport
34//!     let mut buff = [0u8; 256];
35//!     let info = device.request::<AppInfoResp>(AppInfoReq{}, &mut buff, DEFAULT_TIMEOUT).await?;
36//!
37//!     // Await simulator exit or exit signal
38//!     tokio::select!(
39//!         // Await simulator task completion
40//!         _ = driver.wait(&mut handle) => {
41//!             debug!("Complete!");
42//!         }
43//!         // Exit on ctrl + c
44//!         _ = tokio::signal::ctrl_c() => {
45//!             debug!("Exit!");
46//!             driver.exit(handle).await?;
47//!         },
48//!     );
49//!
50//!     Ok(())
51//! }
52//! ```

66/// Device model
67#[derive(Copy, Clone, PartialEq, Debug, VariantNames, Display, EnumString)]
68#[strum(serialize_all = "lowercase")]
69pub enum Model {
70    /// Nano S
71    NanoS,
72    /// Nano S Plus
73    #[strum(serialize = "nanosplus", to_string = "nanosp")]
74    NanoSP,
75    /// Nano X
76    NanoX,
77    /// Stax
78    Stax,
79    /// Flex
80    Flex,
81    /// Nano Gen 5
82    NanoGen5,
83}
84
85impl Model {
86    /// Fetch target name for a given ledger model
87    pub fn target(&self) -> &'static str {
88        match self {
89            Model::NanoS => "nanos",
90            Model::NanoSP => "nanosplus",
91            Model::NanoX => "nanox",
92            Model::Stax => "stax",
93            Model::Flex => "flex",
94            Model::NanoGen5 => "nanogen5",
95        }
96    }
97}

99/// Simulator display mode
100#[derive(Copy, Clone, PartialEq, Debug, VariantNames, Display, EnumString, clap::ValueEnum)]
101#[strum(serialize_all = "lowercase")]
102pub enum Display {

111/// Simulator options
112#[derive(Clone, PartialEq, Debug, Parser)]
113pub struct Options {
114    /// Model to simulate
115    #[clap(long, default_value_t = Options::default().model)]
116    pub model: Model,
117
118    /// Display mode
119    #[clap(long, value_enum)]
120    pub display: Option<Display>,
121
122    /// SDK version override (defaults based on --model)
123    #[clap(long)]
124    pub sdk: Option<String>,
125
126    /// API level override
127    #[clap(long)]
128    pub api_level: Option<String>,
129
130    /// BIP39 seed for initialisation
131    #[clap(long, env)]
132    pub seed: Option<String>,
133
134    /// Enable HTTP API port
135    #[clap(long, default_value_t = Options::default().http_port)]
136    pub http_port: u16,
137
138    /// Enable APDU TCP port (usually 1237)
139    #[clap(long, env)]
140    pub apdu_port: Option<u16>,
141
142    /// Enable debugging and wait for GDB connection (port 1234)
143    #[clap(long)]
144    pub debug: bool,
145
146    /// Speculos root (used to configure python paths if set)
147    #[clap(long, env = "SPECULOS_ROOT")]
148    pub root: Option<String>,
149
150    /// Trace syscalls
151    #[clap(long)]
152    pub trace: bool,
153}
154
155impl Default for Options {
156    fn default() -> Self {
157        Self {
158            model: Model::NanoSP,
159            display: None,
160            sdk: None,

164            apdu_port: None,
165            debug: false,
166            root: None,
167            trace: false,
168        }
169    }
170}

172impl Options {
173    /// Build an argument list from [Options]
174    pub fn args(&self) -> Vec<String> {
175        // Basic args
176        let mut args = vec![
177            format!("--model={}", self.model),
178            format!("--api-port={}", self.http_port),
179        ];
180
181        if let Some(display) = &self.display {
182            args.push(format!("--display={display}"));
183        }
184
185        if let Some(seed) = &self.seed {
186            args.push(format!("--seed={seed}"));
187        }
188
189        if let Some(apdu_port) = &self.apdu_port {
190            args.push(format!("--apdu-port={apdu_port}"));
191        }
192
193        if let Some(sdk) = &self.sdk {
194            args.push(format!("--sdk={sdk}"));
195        }
196
197        if let Some(api_level) = &self.api_level {
198            args.push(format!("--apiLevel={api_level}"));
199        }
200
201        if self.debug {
202            args.push("--debug".to_string());
203        }
204
205        if self.trace {
206            args.push("-t".to_string());
207        }
208
209        args
210    }
211
212    /// Build environmental variable list from [Options]
213    pub fn env(&self) -> HashMap<String, String> {
214        let mut env = HashMap::new();
215
216        if let Some(seed) = &self.seed {
217            env.insert("SPECULOS_SEED".to_string(), seed.clone());
218        }
219
220        env
221    }
222}

224#[cfg(test)]
225mod tests {
226    use std::str::FromStr;

228    use crate::Model;

230    #[test]
231    fn model_name_encoding() {
232        let t = &[
233            (Model::NanoS, "nanos", "nanos"),
234            (Model::NanoSP, "nanosp", "nanosp"),
235            (Model::NanoSP, "nanosp", "nanosplus"),
236            (Model::NanoX, "nanox", "nanox"),
237            (Model::Stax, "stax", "stax"),
238            (Model::Flex, "flex", "flex"),
239            (Model::NanoGen5, "nanogen5", "nanogen5"),
240        ];
241
242        for (model, enc, dec) in t {
243            assert_eq!(&model.to_string(), enc);
244            assert_eq!(Ok(*model), Model::from_str(dec));
245        }
246    }
